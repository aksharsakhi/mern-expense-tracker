\begin{frame}{Architecture, REST APIs \& MongoDB Aggregations}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Three-Tier System Architecture}
        \scriptsize
        \begin{itemize}
          \item \textbf{Frontend (React 18 + Vite):} Glassmorphism UI, Chart.js visuals.
          \item \textbf{Backend (Node.js + Express):} RESTful API, CORS, input validators.
          \item \textbf{Database (MongoDB 8.x):} Compound indexes on \texttt{(date, type)}.
        \end{itemize}
      \end{block}

      \begin{exampleblock}{RESTful Endpoints Contract}
        \scriptsize
        \begin{itemize}
          \item \texttt{GET  /api/transactions?month=2026-10} (Filtered list)
          \item \texttt{GET  /api/transactions/summary} (MongoDB aggregation)
          \item \texttt{POST /api/transactions} (Create record with validation)
          \item \texttt{PUT /api/transactions/:id} \& \texttt{DELETE} (Update/Remove)
          \item \texttt{POST /api/transactions/seed} (One-click demo dataset)
        \end{itemize}
      \end{exampleblock}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{Multi-Faceted MongoDB Aggregation}
        \scriptsize
        Single round-trip database query utilizing \texttt{\$facet}:
        \begin{itemize}
          \item \textbf{Totals Facet:} Computes total income, expenses, count.
          \item \textbf{Category Distribution:} Group expenditures to calculate percentage share per category for donut chart.
          \item \textbf{Daily Cashflow Velocity:} Group daily spending to feed dynamic bar chart.
        \end{itemize}
      \end{block}

      \begin{exampleblock}{Computed Real-Time Metrics}
        \scriptsize
        \begin{itemize}
          \item $\text{\textbf{Net Balance}} = \text{Total Income} - \text{Total Expense}$
          \item $\text{\textbf{Savings Rate}} = \frac{\text{Income} - \text{Expense}}{\text{Income}} \times 100\%$
          \item \textbf{Payment Splits:} Breakdown across UPI, Cards, Net Banking.
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}
