\begin{frame}{Problem Statement \& Motivation}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{The Challenge}
        \begin{itemize}
          \item Personal financial records are fragmented across static notes, paper bills, and disjointed bank statements.
          \item Lack of real-time insights into monthly category distributions (e.g., dining, housing, transportation).
          \item Absence of instant metrics for true monthly net balance and dynamic savings velocity.
        \end{itemize}
      \end{alertblock}
    \end{column}
    
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{The Solution: SpendWise}
        \begin{itemize}
          \item Centralized full-stack web application built on the modern \textbf{MERN} stack.
          \item Fast transaction logging with payment methods, category tagging, and date tracking.
          \item High-speed aggregation pipelines calculating cashflow statistics with visual Chart.js dashboards.
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

% SLIDE 5: MONGODB AGGREGATION PIPELINES
% -------------------------------------------------------------
\begin{frame}{MongoDB Aggregation Pipelines}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Multi-Faceted Pipeline Execution}
        A single round-trip database query utilizing \texttt{\$facet}:
        \begin{enumerate}
          \item \textbf{Total Balances:} \texttt{\$group} by \texttt{\$type} to sum total income and total expense.
          \item \textbf{Category Distribution:} Filter expenses and compute sub-totals and percentages per category.
          \item \textbf{Daily Cashflow Velocity:} Group by ISO date string to feed daily spending charts.
        \end{enumerate}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Computed Metrics Returned}
        \begin{itemize}
          \item \textbf{Net Balance:} $\text{Total Income} - \text{Total Expense}$
          \item \textbf{Savings Rate:} $\frac{\text{Income} - \text{Expense}}{\text{Income}} \times 100\%$
          \item \textbf{Category Share:} $\frac{\text{Category Total}}{\text{Total Expense}} \times 100\%$
          \item \textbf{Payment Method Split:} Breakdown across UPI, Cards, and Net Banking.
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Production Deployment \& DevOps Architecture}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Containerization (\texttt{Docker})}
        \begin{itemize}
          \item \textbf{Backend:} Multi-stage Node.js Alpine container with non-root security.
          \item \textbf{Frontend:} Multi-stage build producing static bundle served through an optimized \textbf{Nginx} Alpine reverse proxy.
          \item \textbf{Compose:} \texttt{docker-compose.yml} orchestrating MongoDB, Express API, and Nginx Client.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Cloud Infrastructure as Code}
        \begin{itemize}
          \item \textbf{Render Blueprint (\texttt{render.yaml}):} Automated cloud deployment for Express web service and React static site.
          \item \textbf{Vercel (\texttt{vercel.json}):} Client-side SPA routing and transparent API proxying.
          \item \textbf{Hybrid Production Serving:} Express dynamically serves \texttt{client/dist} in production single-instance environments.
        \end{itemize}
